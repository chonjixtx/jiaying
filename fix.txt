# balls 一键删除已安装坏包命令 —— 在 NewTerm 里以 root 粘贴执行（不要复制这一行 # 注释）
# 这条命令只删记录，不再安装任何东西，也不会触发 dpkg 维护脚本，不会黑屏。
D=$(command -v dpkg 2>/dev/null); [ -x "$D" ] || D=/var/jb/usr/bin/dpkg; J=$(dirname "$(dirname "$D")"); R="$J/var/lib/dpkg"; rm -f "$R"/info/com.jiaying.gamehall.* "$R"/info/com.balls.cleanup* "$R"/info/com.balls.mapping* "$R"/updates/*; cp "$R/status" "$R/status.ballsbak"; awk 'BEGIN{RS="";ORS="\n\n"} {split($0,L,"\n"); if(L[1]~/^Package: (com\.jiaying\.gamehall|com\.balls\.(cleanup|cleanup2|mapping|mapping2))$/)next; print}' "$R/status" > "$R/status.ballsnew"; cat "$R/status.ballsnew" > "$R/status"; rm -rf "$J/Applications/JiayingGameHall.app"; rm -f "$J/Library/MobileSubstrate/DynamicLibraries/JiayingGameTweak."* "$J/usr/lib/TweakInject/JiayingGameTweak."*; uicache -r; echo BALLS_CLEANED
